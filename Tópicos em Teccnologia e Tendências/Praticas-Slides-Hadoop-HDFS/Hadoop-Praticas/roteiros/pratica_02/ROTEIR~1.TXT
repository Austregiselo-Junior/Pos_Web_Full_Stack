# 3) Criar estrutura via beeline:

    [cloudera@sandbox ~]$ beeline

    beeline> !connect jdbc:hive2://sandbox-hdp.hortonworks.com:2181/;serviceDiscoveryMode=zooKeeper;zooKeeperNamespace=hiveserver2
    [ENTER]
    [ENTER]
    beeline> show databases;
    beeline> describe database stg_treinamento_coii;
    beeline> describe database treinamento_coii;
    beeline> !run /home/cloudera/scripts_ricardo/treinamento_coii/deploy/teste_hdp_external.sql
    beeline> !run /home/cloudera/scripts_ricardo/treinamento_coii/deploy/teste_hdp_orc.sql
    beeline> use stg_treinamento_coii;
    beeline> show tables;
    beeline> select * from ex_teste_hdp;
    beeline> use treinamento_coii;
    beeline> show tables;
    beeline> select * from teste_hdp;
    beeline> !exit

# 4) Put dos dados da tabela 'ex_teste_hdp' para o HDFS

    [cloudera@sandbox ~]$ hdfs dfs -put -f /home/cloudera/scripts_ricardo/treinamento_coii/dados/* /dados/staging_area/tst/treinamento_coii/
