    >>> from pyspark.sql.types import *

    # 2.4) Transformações com Dataframes parte 1: Schemas, filtros, agrupamentos, planos de execução e o Pyspark deduzindo schemas de arquivo json
    
    >>> peopleDF = spark.read.json("C:/bigdata/datasets/people.json")
    >>> peopleDF.printSchema()

    >>> peopleDF.filter(peopleDF.age > 25).explain()
    >>> peopleDF.filter(peopleDF.age > 25).show()
    >>> peopleDF.filter("age > 25").show()

    >>> peopleDF.filter((peopleDF.age > 25) & (peopleDF.age <= 50)).show()
    >>> peopleDF.filter(peopleDF.gender == "F").show()
    >>> peopleDF.groupBy(peopleDF.gender).count().explain()
    >>> peopleDF.filter(peopleDF.age>25).orderBy(peopleDF.age).explain()
    >>> peopleDF.groupBy(peopleDF.gender).count().show()

    >>> peopleDF.createOrReplaceTempView("people") 
    >>> spark.sql("SELECT name, age FROM people WHERE gender = 'F' and age = (select max(age) FROM people WHERE gender='F')").show()

    # 2.5) Trabalhando com Dataframes mais complexos

    >>> githubDF = spark.read.json("C:/bigdata/datasets/github.json")
    >>> githubDF.limit(5).show()

    >>> githubDF.limit(5).select(githubDF.actor).show(truncate=False)
    >>> githubDF.select("actor").printSchema()

    >>> actorDF = githubDF.select("actor")
    >>> actorDF.select("actor.login").limit(5).show()

    // Contando logins únicos
    >>> actorDF.select("actor.login").dropDuplicates().count()

    // Contando todos os valores únicos da coluna do tipo 'type'.
    >>> githubDF.dropDuplicates(["type"]).count()
    >>> githubDF.dropDuplicates(["type"]).show()

    //  Quantidade dos registros do tipo "CreateEvent"
    >>> githubDF.filter(githubDF.type == "CreateEvent").count()

    // GroupBy por login de ator  e seu respectivo schema
    >>> githubDF.groupBy("actor.login").count().printSchema()
    >>> githubDF.groupBy("actor.login").count().limit(5).show()

    // Exiba os 20 logins com o maior número de contribuições
    >>> githubDF.groupBy("actor.login").count().orderBy("count", ascending=False).limit(20).show()

    // Opcional: Usando SQL para a consulta anterior

    // Criando tabela temporária
    >>> githubDF.createOrReplaceTempView("githubData")

    // Executando a Query
    >>> spark.sql("SELECT actor.login, COUNT(*) AS entryCount FROM githubData GROUP BY actor.login ORDER BY entryCount DESC LIMIT 20").show()

# 3) Cache, Joins e Broadcast

    // Cache
    >>> githubDF.count()
    >>> githubDF.cache()
    >>> githubDF.count()
    >>> githubDF.count()
    >>> githubDF.count()
    
    // Top 20 usuários que mais contribuíram no Github: Sem Join

    >>> scanQuery = githubDF.groupBy("actor.login").count().orderBy("count", ascending=False).limit(20)
    >>> scanQuery.show()

    // Top 20 usuários que mais contribuíram no Github: Com Join + Broadcast

    >>> githubTop20DF = spark.read.json("C:/bigdata/datasets/github-top20.json")
    >>> githubTop20DF.show()
    >>> topContributorsJoinedDF = githubDF.join(githubTop20DF, githubDF.actor.login == githubTop20DF.login).groupBy(githubDF.actor.login).count().orderBy("count", ascending=False)
    
    // Shuffle realizado > 7900x
    >>> topContributorsJoinedDF.show()

    // Plano de Execução
    >>> topContributorsJoinedDF.explain()

    // Cenário sem broadcast
    >>> spark.conf.set("spark.sql.autoBroadcastJoinThreshold",-1)

    // Execute novamente o Join e execute o explain sem broadcast  
    >>> topContributorsJoinedDF = githubDF.join(githubTop20DF, githubDF.actor.login == githubTop20DF.login).groupBy(githubDF.actor.login).count().orderBy("count", ascending=False)
    >>> topContributorsJoinedDF.explain()
    >>> topContributorsJoinedDF.show()

    // Ativando novamente o broadcast para o valor default de 10MB
    >>> spark.conf.set("spark.sql.autoBroadcastJoinThreshold",1024*1024*10)
