from pyspark.sql import HiveContext
contexto = HiveContext(sc)

banco = contexto.table("default.products")
banco.show()
banco.registerTempTable("products")

contexto.sql("select * from products").show()
contexto.sql("select sum(product_price) from products").show()

vendas = contexto.sql(" select * from products ")
vendas.show()

vendas.show(100)
vendas.printSchema()

vendas.select('product_name','product_price').show()

vendas.select('product_name','product_price').distinct().show(30)

vendas.filter(vendas.product_id==3).count()
